cmake_minimum_required(VERSION 3.16)
project(tm_imu_orientation_filter VERSION 0.1.0 LANGUAGES C)
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(TM_GYRO_METHOD "FAST" CACHE STRING "Default gyroscope method: FAST or EXACT")
set_property(CACHE TM_GYRO_METHOD PROPERTY STRINGS FAST EXACT)
option(TM_BUILD_VIEWER "Build the POSIX raylib serial viewer" OFF)

add_library(tm_filter src/tm_filter.c)
target_include_directories(tm_filter PUBLIC include)
if(TM_GYRO_METHOD STREQUAL "EXACT")
  target_compile_definitions(tm_filter PUBLIC TM_GYRO_METHOD=TM_GYRO_METHOD_EXACT)
elseif(NOT TM_GYRO_METHOD STREQUAL "FAST")
  message(FATAL_ERROR "TM_GYRO_METHOD must be FAST or EXACT")
endif()
if(NOT MSVC)
  target_compile_options(tm_filter PRIVATE -Wall -Wextra -Wpedantic)
  target_link_libraries(tm_filter PUBLIC m)
endif()
add_executable(basic_example examples/basic_example.c)
target_link_libraries(basic_example PRIVATE tm_filter)

include(CTest)
if(BUILD_TESTING)
  add_executable(test_filter tests/test_filter.c examples/desktop/madgwick.c examples/desktop/imu_packet.c)
  target_include_directories(test_filter PRIVATE examples/desktop)
  target_link_libraries(test_filter PRIVATE tm_filter)
  add_test(NAME filter_math_and_serial COMMAND test_filter)
endif()
if(TM_BUILD_VIEWER)
  if(NOT UNIX)
    message(FATAL_ERROR "The serial viewer requires macOS or Linux; the core filter is portable C99.")
  endif()
  find_package(raylib QUIET)
  if(NOT raylib_FOUND)
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(RAYLIB REQUIRED IMPORTED_TARGET raylib)
    set(TM_RAYLIB_TARGET PkgConfig::RAYLIB)
  else()
    set(TM_RAYLIB_TARGET raylib)
  endif()
  add_executable(imu_viewer examples/desktop/imu_viewer.c examples/desktop/madgwick.c examples/desktop/imu_packet.c)
  target_link_libraries(imu_viewer PRIVATE tm_filter ${TM_RAYLIB_TARGET})
endif()
